-- 1. Table for Friday Parlays
CREATE TABLE IF NOT EXISTS friday_parlays (
  id TEXT PRIMARY KEY,
  week_date DATE NOT NULL,
  bettor_name TEXT NOT NULL,
  status TEXT DEFAULT 'building',
  stake NUMERIC DEFAULT 25.00,
  total_odds TEXT,
  potential_payout NUMERIC,
  notes TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 2. Table for Individual Boys' Legs
CREATE TABLE IF NOT EXISTS parlay_legs (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  week_date DATE NOT NULL,
  user_name TEXT NOT NULL,
  game_title TEXT NOT NULL,
  pick_text TEXT NOT NULL,
  odds TEXT NOT NULL,
  bet_type TEXT,
  chirp TEXT,
  locked BOOLEAN DEFAULT TRUE,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  UNIQUE(week_date, user_name)
);

-- 3. Enable RLS and public policies for the boys
ALTER TABLE friday_parlays ENABLE ROW LEVEL SECURITY;
ALTER TABLE parlay_legs ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Allow public all" ON friday_parlays FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Allow public all" ON parlay_legs FOR ALL USING (true) WITH CHECK (true);

-- 4. Enable Realtime updates
ALTER PUBLICATION supabase_realtime ADD TABLE friday_parlays;
ALTER PUBLICATION supabase_realtime ADD TABLE parlay_legs;
